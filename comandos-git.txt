ssh-keygen -t ed25519 -C email@gmail.com - Gera o id_ed25519
*Copie o conteudo da chave pública e depois colo no github.
eval $(ssh-agent -s) - Inicializa o ssh agent.
ssh-add [caminho da chava privada] - Passa a chave privada para o agente ssh.

git init
git config --global user.email "email@gmail.com"
git config --global user.name "nome"

git add [nome do arquivo] - Adiciona somente os arquivos selecionados.
git add . - Adiciona todas a modificações ao staging area.
git add * - Adiciona somente as modificações do diretário atual.
git commit -m "Mensagem do commit"

git status - Mostra o status do working tree.

git config --list - Lista todas as configurações.
git config --global --unset [configuração] - Apaga essa configuração.

git remote add origin [Caminho do repositório remoto] - Empurra o repositório local ao repositório remoto.
git remote -v - Mostra uma lista de repositórios remotos cadastrados.
git push -u origin[apelido] main - Empurra a branch main local para o remoto.
git branch -M main - Renomeia a brach atual por "main"
git pull
git log --stat - Lista todos os commits já feitos.
git rebase -i <hash do commit> - Altera um commit anterior
git config --global credential.helper 'cache --timeout=3600' -- guarda o token pessoal por 3600 segundos.
git config --global credential.helper store - Guarda permanentemente no arquivo ~/.git-credentials
